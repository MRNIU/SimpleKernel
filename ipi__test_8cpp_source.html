24// ===========================================================================
25// IPI handler for delivery verification tests.
26// Performs original handler duties (clear SSIP on riscv64) + increments
27// the shared counter so the test thread can observe receipt.
28// ===========================================================================
29
30auto TestIpiHandler(uint64_t, cpu_io::TrapContext*) -> uint64_t {
31#if defined(__riscv)
32 cpu_io::Sip::Ssip::Clear();
33#endif
34 g_ipi_received_count.fetch_add(1, std::memory_order_release);
35 return 0;
36}

38auto OriginalIpiHandler(uint64_t, cpu_io::TrapContext*) -> uint64_t {
