52extern "C" auto __cxa_atexit(void (*destructor_func)(void*), void* obj_ptr,
53 void*) -> int {
54 if (atexit_func_count >= kMaxAtExitFuncsCount) {
55 return -1;
56 }
57 atexit_funcs[atexit_func_count].destructor_func = destructor_func;
58 atexit_funcs[atexit_func_count].obj_ptr = obj_ptr;
59 atexit_funcs[atexit_func_count].dso_handle = nullptr;
60 atexit_func_count++;
61 return 0;

69extern "C" auto __cxa_finalize(void* destructor_func) -> void {
70 if (destructor_func == nullptr) {
71 // 如果 destructor_func 为 nullptr，调用所有析构函数
72 for (size_t i = atexit_func_count; i > 0; --i) {
73 size_t idx = i - 1;
74 if (atexit_funcs[idx].destructor_func != nullptr) {
75 (*atexit_funcs[idx].destructor_func)(atexit_funcs[idx].obj_ptr);
76 }
77 }
78 } else {

84 (*atexit_funcs[idx].destructor_func)(atexit_funcs[idx].obj_ptr);

128extern "C" auto __cxa_guard_acquire(GuardType* guard) -> int {
129 uint64_t expected = 0;
130
131 // 紧凑的 CAS 循环：尝试将 0 -> kInUseMask
132 // compare_exchange_weak 失败时会将当前值写入 expected
133 while (!guard->guard.compare_exchange_weak(expected, GuardType::kInUseMask,
134 std::memory_order_acq_rel,
135 std::memory_order_acquire)) {
136 // CAS 失败，检查失败原因
137 if ((expected & GuardType::kInitializedMask) != 0U) {
138 // 已被其他核心初始化完成
139 return 0;
140 }
141
142 // 其他核心正在初始化中，等待
143 cpu_io::Pause();
144
145 // 重置 expected 以便下次 CAS 尝试
146 // 如果当前值仍有 kInUseMask，CAS 会再次失败并更新 expected
147 expected = 0;
148 }
149
150 // CAS 成功，当前核心负责初始化
151 return 1;
152}
