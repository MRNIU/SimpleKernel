13auto TaskManager::SendSignal(Pid pid, int signum) -> Expected<void> {
14 if (!SignalState::IsValid(signum)) {
15 return std::unexpected(Error{ErrorCode::kSignalInvalidNumber});
16 }
17
18 // 查找目标任务
19 TaskControlBlock* target = nullptr;
20 {
21 LockGuard<SpinLock> lock_guard(task_table_lock_);
22 auto it = task_table_.find(pid);
23 if (it == task_table_.end()) {
24 return std::unexpected(Error{ErrorCode::kSignalTaskNotFound});
25 }
26 target = it->second.get();
27 }
28
29 if (!target || !target->aux) {
30 return std::unexpected(Error{ErrorCode::kSignalTaskNotFound});
31 }
32
33 target->aux->signals.SetPending(signum);
34
35 klog::Debug("SendSignal: signal {} ({}) sent to pid={}",
36 SignalState::GetSignalName(signum), signum, pid);
37
38 // SIGKILL 和 SIGCONT 需要立即唤醒阻塞/睡眠的任务
39 bool needs_wake =
40 (signum == signal_number::kSigKill || signum == signal_number::kSigCont);
41
42 // 如果信号未被屏蔽，也需要唤醒
43 if (!needs_wake &&
44 !(target->aux->signals.blocked.load(std::memory_order_acquire) &
45 (1U << signum))) {
46 auto status = target->GetStatus();
47 if (status == TaskStatus::kBlocked || status == TaskStatus::kSleeping) {
48 needs_wake = true;
49 }
50 }
51
52 if (needs_wake) {
53 auto status = target->GetStatus();
54
55 if (status == TaskStatus::kBlocked) {
56 auto resource_id = target->aux->blocked_on;
57 if (static_cast<bool>(resource_id)) {
58 Wakeup(resource_id);
59 klog::Debug("SendSignal: woke blocked task pid={} for signal {}", pid,
60 SignalState::GetSignalName(signum));
61 }
62 } else if (status == TaskStatus::kSleeping) {
63 // Expedite wake: set wake_tick to 0 so next TickUpdate wakes it.
64 // Note: single aligned 64-bit store is atomic on 64-bit platforms.
65 target->sched_info.wake_tick = 0;
66 klog::Debug("SendSignal: expedited sleep wakeup for pid={}", pid);
67 }
68 }
69
70 return {};
71}

73auto TaskManager::CheckPendingSignals() -> int {
74 auto* current = GetCurrentTask();
75 if (!current || !current->aux) {
76 return 0;
77 }
78
79 auto& signals = current->aux->signals;
80 if (!signals.HasDeliverableSignal()) {
81 return 0;
82 }
83
84 int signum = signals.GetNextDeliverableSignal();
85 if (signum == signal_number::kSigNone) {
86 return 0;
87 }
88
89 signals.ClearPending(signum);
90
91 klog::Debug("CheckPendingSignals: pid={} delivering signal {} ({})",
92 current->pid, signum, SignalState::GetSignalName(signum));
93
94 auto& action = signals.actions[signum];
95
96 // SIGKILL: 强制终止，无法捕获
97 if (signum == signal_number::kSigKill) {
98 klog::Info("CheckPendingSignals: pid={} killed by SIGKILL", current->pid);
99 Exit(128 + signum);
100 __builtin_unreachable();
101 }
102
103 // SIGSTOP: 强制停止，无法捕获
105 if (signum == signal_number::kSigStop || signum == signal_number::kSigTstp) {
106 klog::Info(
107 "CheckPendingSignals: pid={} received stop signal {} (not implemented)",
108 current->pid, SignalState::GetSignalName(signum));
109 return signum;
110 }
111
112 // SIGCONT: 继续（如果已停止）
113 if (signum == signal_number::kSigCont) {
114 klog::Debug("CheckPendingSignals: pid={} received SIGCONT", current->pid);
115 return signum;
116 }
117
118 if (action.handler == kSigIgn) {
119 klog::Debug("CheckPendingSignals: pid={} ignored signal {}", current->pid,
120 SignalState::GetSignalName(signum));
121 return signum;
122 }
123
124 if (action.handler == kSigDfl) {
125 char default_action = GetDefaultSignalAction(signum);
126 switch (default_action) {
127 case 'T':
128 case 'C':
129 klog::Info(
130 "CheckPendingSignals: pid={} terminated by signal {} (default)",
131 current->pid, SignalState::GetSignalName(signum));
132 Exit(128 + signum);
133 __builtin_unreachable();
134 case 'I':
135 break;
136 case 'S':
137 klog::Debug("CheckPendingSignals: pid={} stop by {} (not implemented)",
138 current->pid, SignalState::GetSignalName(signum));
139 break;
140 case 'K':
141 break;
142 default:
143 break;
144 }
145 return signum;
146 }
147
154 klog::Debug(
155 "CheckPendingSignals: pid={} user handler for signal {} (delivery "
156 "not yet implemented)",
157 current->pid, SignalState::GetSignalName(signum));
158 return signum;
159}

190auto TaskManager::SetSignalMask(int how, uint32_t set, uint32_t* oldset)
191 -> Expected<void> {
192 auto* current = GetCurrentTask();
193 if (!current || !current->aux) {
194 return std::unexpected(Error{ErrorCode::kTaskNoCurrentTask});
195 }
196
197 auto& signals = current->aux->signals;
198
199 if (oldset) {
200 *oldset = signals.blocked.load(std::memory_order_acquire);
201 }
202
203 // SIGKILL 和 SIGSTOP 不能被屏蔽
204 uint32_t uncatchable_mask =
205 (1U << signal_number::kSigKill) | (1U << signal_number::kSigStop);
206 set &= ~uncatchable_mask;
207
208 switch (how) {
209 case signal_mask_op::kSigBlock:
210 signals.blocked.fetch_or(set, std::memory_order_release);
211 break;
212 case signal_mask_op::kSigUnblock:
213 signals.blocked.fetch_and(~set, std::memory_order_release);
214 break;
215 case signal_mask_op::kSigSetmask:
216 signals.blocked.store(set, std::memory_order_release);
217 break;
218 default:
219 return std::unexpected(Error{ErrorCode::kInvalidArgument});
220 }
221
222 signals.blocked.fetch_and(~uncatchable_mask, std::memory_order_release);
223
224 klog::Debug("SetSignalMask: pid={} mask={:#x}", current->pid,
225 signals.blocked.load(std::memory_order_acquire));
226
227 return {};
228}
