34class BmallocLock : public bmalloc::LockBase {
35 public:
36 void Lock() override {
37 saved_intr_ = cpu_io::GetInterruptStatus();
38 cpu_io::DisableInterrupt();
39
40 lock_.Lock().or_else([this](auto&&) -> Expected<void> {
41 // 不应触发：bmalloc 内部不会递归加锁
42 // 恢复中断状态后再死循环，避免关中断挂死整个核心
43 if (saved_intr_) {
44 cpu_io::EnableInterrupt();
45 }
46 while (true) {
47 cpu_io::Pause();
48 }
49 return {};
50 });
51 }
52
53 void Unlock() override {
54 lock_.UnLock().or_else([](auto&&) -> Expected<void> {
55 while (true) {
56 cpu_io::Pause();
57 }
58 return {};
59 });
60
61 if (saved_intr_) {
62 cpu_io::EnableInterrupt();
63 }
64 }
65
66 private:
67 SpinLock lock_{"bmalloc"};
68 bool saved_intr_{false};
69};
