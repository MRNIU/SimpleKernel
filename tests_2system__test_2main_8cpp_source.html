130void test_runner_entry(void* /*arg*/) {
131 auto& task_mgr = TaskManagerSingleton::instance();
132 auto* runner = task_mgr.GetCurrentTask();
133
134 // Phase 2: 为每个非 SMP 测试创建独立线程，建立父子关系
135 int thread_test_count = 0;
136
137 for (size_t i = 0; i < kTestCount; ++i) {
138 if (test_cases[i].is_smp_test) {
139 continue;
140 }
141
142 auto task = kstd::make_unique<TaskControlBlock>(
143 test_cases[i].name, 10, test_thread_entry, reinterpret_cast<void*>(i));
144
145 // Wait() requires parent-child relationship to collect exit status
146 task->aux->parent_pid = runner->pid;
147 task->aux->pgid = runner->aux->pgid;
148
149 // Save raw pointer: pid is assigned inside AddTask() by AllocatePid(),
150 // so we must read it *after* the call (unique_ptr is moved).
151 auto* task_ptr = task.get();
152 test_results[i].status = TestThreadStatus::kRunning;
153
154 task_mgr.AddTask(std::move(task));
155 test_results[i].pid = task_ptr->pid;
156 thread_test_count++;
157 }
158
159 klog::Info("[RUNNER] Spawned {} test threads, collecting via Wait()...",
160 thread_test_count);
161
162 // Phase 3: 通过 Wait() 收集所有测试线程的退出状态
163 int collected = 0;
164 constexpr int kMaxWaitRetries = 1200; // 1200 * 50ms = 60s 超时
165 int retries = 0;
166
167 while (collected < thread_test_count && retries < kMaxWaitRetries) {
168 int status = 0;
169 auto wait_result =
170 task_mgr.Wait(static_cast<Pid>(-1), &status, true, false);
171
172 if (wait_result.has_value() && wait_result.value() > 0) {
173 Pid exited_pid = wait_result.value();
174
175 bool is_test_thread = false;
176 for (size_t i = 0; i < kTestCount; ++i) {
177 if (test_results[i].pid == static_cast<int64_t>(exited_pid)) {
178 test_results[i].exit_code = status;
179 test_results[i].status = (status == 0) ? TestThreadStatus::kPassed
180 : TestThreadStatus::kFailed;
181
182 klog::Info("[RUNNER] Collected: {} (pid={}, exit_code={}) — {}",
183 test_cases[i].name, exited_pid, status,
184 (status == 0) ? "PASS" : "FAIL");
185 is_test_thread = true;
186 break;
187 }
188 }
189
190 if (is_test_thread) {
191 collected++;
192 } else {
193 klog::Debug("[RUNNER] Reaped orphan pid={}, not a test thread",
194 exited_pid);
195 }
196 } else {
197 (void)sys_sleep(50);
198 retries++;
199 }
200 }
201
202 for (size_t i = 0; i < kTestCount; ++i) {
203 if (test_cases[i].is_smp_test) {
204 continue;
205 }
206 if (test_results[i].status == TestThreadStatus::kPending ||
207 test_results[i].status == TestThreadStatus::kRunning) {
208 test_results[i].status = TestThreadStatus::kTimeout;
209 klog::Err("[RUNNER] Timeout: {} (pid={})", test_cases[i].name,
210 test_results[i].pid);
211 }
212 }
213
214 // Phase 4: 汇总打印
215 print_test_summary();
216
217 bool all_passed = true;
218 for (size_t i = 0; i < kTestCount; ++i) {
219 if (test_results[i].status != TestThreadStatus::kPassed) {
220 all_passed = false;
221 break;
222 }
223 }
224
225 QemuExit(all_passed);
226}
