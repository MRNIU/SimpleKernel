122auto TaskManager::AddTask(etl::unique_ptr<TaskControlBlock> task) -> void {
123 assert(task.get() != nullptr && "AddTask: task must not be null");
124 assert(task->GetStatus() == TaskStatus::kUnInit &&
125 "AddTask: task status must be kUnInit");
126 // 分配 PID
127 if (task->pid == 0) {
128 task->pid = AllocatePid();
129 }
130
131 // 如果 tgid 未设置，则将其设为自己的 pid (单线程进程或线程组的主线程)
132 if (task->aux->tgid == 0) {
133 task->aux->tgid = task->pid;
134 }
135
136 auto* task_ptr = task.get();
137 Pid pid = task_ptr->pid;
138
139 // 确定目标核心
140 auto target_core = cpu_io::GetCurrentCoreId();
141 if (task_ptr->aux->cpu_affinity.value() != UINT64_MAX) {
142 // 寻找第一个允许的核心
143 for (size_t core_id = 0; core_id < SIMPLEKERNEL_MAX_CORE_COUNT; ++core_id) {
144 if (task_ptr->aux->cpu_affinity.value() & (1UL << core_id)) {
145 target_core = core_id;
146 break;
147 }
148 }
149 }
150
151 // 设置任务状态为 kReady（task 尚未对其他核心可见，无需锁）
152 // Transition: kUnInit -> kReady
153 task_ptr->fsm.Receive(MsgSchedule{});
154
155 auto& cpu_sched = cpu_schedulers_[target_core];
156
157 // 锁序统一为 sched_lock → task_table_lock（与 Exit/Wait 一致，防止死锁）
158 {
159 LockGuard<SpinLock> sched_guard(cpu_sched.lock);
160 {
161 LockGuard<SpinLock> table_guard(task_table_lock_);
162 if (task_table_.full()) {
163 klog::Err("AddTask: task_table_ full, cannot add task (pid={})", pid);
164 return;
165 }
166 task_table_[pid] = std::move(task);
167 }
168
169 assert(task_ptr->policy < SchedPolicy::kPolicyCount &&
170 "AddTask: invalid scheduling policy");
171 if (cpu_sched.schedulers[static_cast<uint8_t>(task_ptr->policy)]) {
172 cpu_sched.schedulers[static_cast<uint8_t>(task_ptr->policy)]->Enqueue(
173 task_ptr);
174 }
175 }
176}

194auto TaskManager::ReapTask(TaskControlBlock* task) -> void {
195 if (!task) {
196 return;
197 }
198
199 // 确保任务处于僵尸或退出状态
200 if (task->GetStatus() != TaskStatus::kZombie &&
201 task->GetStatus() != TaskStatus::kExited) {
202 klog::Warn("ReapTask: Task {} is not in zombie/exited state", task->pid);
203 return;
204 }
205
206 // Capture pid before erase (unique_ptr deletes on erase)
207 Pid pid = task->pid;
208
209 // 从全局任务表中移除 (unique_ptr auto-deletes TCB)
210 {
211 LockGuard lock_guard{task_table_lock_};
212 task_table_.erase(pid);
213 }
214
215 klog::Debug("ReapTask: Task {} resources freed", pid);
216}

218auto TaskManager::ReparentChildren(TaskControlBlock* parent) -> void {
219 if (!parent) {
220 return;
221 }
222
223 LockGuard lock_guard{task_table_lock_};
224
225 // 遍历所有任务，找到父进程是当前任务的子进程
226 for (auto& [pid, task] : task_table_) {
227 if (task && task->aux->parent_pid == parent->pid) {
228 // 将子进程过继给 init 进程
229 task->aux->parent_pid = kInitPid;
230 klog::Debug("ReparentChildren: Task {} reparented to init (PID {})",
231 task->pid, kInitPid);
232 }
