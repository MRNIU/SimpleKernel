42 auto& cpu_sched = cpu_schedulers_[core_id];
43
44 LockGuard lock_guard{cpu_sched.lock};
45
46 if (!cpu_sched.schedulers[static_cast<uint8_t>(SchedPolicy::kNormal)]) {
47 cpu_sched.schedulers[static_cast<uint8_t>(SchedPolicy::kRealTime)] =
48 kstd::make_unique<FifoScheduler>();
49 cpu_sched.schedulers[static_cast<uint8_t>(SchedPolicy::kNormal)] =
50 kstd::make_unique<RoundRobinScheduler>();
51 cpu_sched.schedulers[static_cast<uint8_t>(SchedPolicy::kIdle)] =
52 kstd::make_unique<IdleScheduler>();
53 }
54

56 cpu_data.sched_data = &cpu_sched;
57
58 if (is_primary) {
59 auto init_task_ptr =
60 kstd::make_unique<TaskControlBlock>("init", 10, nullptr, nullptr);
61 auto* init_task = init_task_ptr.get();
62 init_task->pid = kInitPid;
63 init_task->aux->tgid = kInitPid;
64 init_task->policy = SchedPolicy::kNormal;
65 init_task->fsm.Receive(MsgSchedule{});
66 init_task->fsm.Receive(MsgSchedule{});
67
68 {
69 LockGuard<SpinLock> table_guard(task_table_lock_);
70 task_table_[init_task->pid] = std::move(init_task_ptr);
71 }
72
73 cpu_data.running_task = init_task;
74 } else {
75 auto boot_task_ptr = kstd::make_unique<TaskControlBlock>(
76 "boot",
77 std::numeric_limits<
78 decltype(TaskControlBlock::SchedInfo::priority)>::max(),
79 nullptr, nullptr);

81 boot_task->fsm.Receive(MsgSchedule{});
82 boot_task->fsm.Receive(MsgSchedule{});
83 boot_task->policy = SchedPolicy::kIdle;
84 cpu_data.running_task = boot_task;

88 "idle",

104auto TaskManager::AddTask(etl::unique_ptr<TaskControlBlock> task) -> void {
105 assert(task.get() != nullptr && "AddTask: task must not be null");
106 assert(task->GetStatus() == TaskStatus::kUnInit &&
107 "AddTask: task status must be kUnInit");
108 // 分配 PID
109 if (task->pid == 0) {
110 task->pid = AllocatePid();
111 }
112
113 // 如果 tgid 未设置，则将其设为自己的 pid (单线程进程或线程组的主线程)
114 if (task->aux->tgid == 0) {
115 task->aux->tgid = task->pid;
116 }
117
118 auto* task_ptr = task.get();
119 Pid pid = task_ptr->pid;
120
121 // 确定目标核心
122 auto target_core = cpu_io::GetCurrentCoreId();
123 if (task_ptr->aux->cpu_affinity.value() != UINT64_MAX) {
124 // 寻找第一个允许的核心
125 for (size_t core_id = 0; core_id < SIMPLEKERNEL_MAX_CORE_COUNT; ++core_id) {
126 if (task_ptr->aux->cpu_affinity.value() & (1UL << core_id)) {
127 target_core = core_id;
128 break;
129 }
130 }
131 }
132
133 // 设置任务状态为 kReady（task 尚未对其他核心可见，无需锁）
134 // Transition: kUnInit -> kReady
135 task_ptr->fsm.Receive(MsgSchedule{});
136
137 auto& cpu_sched = cpu_schedulers_[target_core];
138
139 // 锁序统一为 sched_lock → task_table_lock（与 Exit/Wait 一致，防止死锁）
140 {
141 LockGuard<SpinLock> sched_guard(cpu_sched.lock);
142 {
143 LockGuard<SpinLock> table_guard(task_table_lock_);
144 if (task_table_.full()) {
145 klog::Err("AddTask: task_table_ full, cannot add task (pid={})", pid);
146 return;
147 }
148 task_table_[pid] = std::move(task);
149 }
150
151 assert(task_ptr->policy < SchedPolicy::kPolicyCount &&
152 "AddTask: invalid scheduling policy");
153 if (cpu_sched.schedulers[static_cast<uint8_t>(task_ptr->policy)]) {
154 cpu_sched.schedulers[static_cast<uint8_t>(task_ptr->policy)]->Enqueue(
155 task_ptr);
156 }
157 }
158}

176auto TaskManager::Balance() -> void {
177 auto current_core = cpu_io::GetCurrentCoreId();
178 auto& current_sched = cpu_schedulers_[current_core];
179
180 // 获取当前核心 kNormal 队列长度（无锁快速检查）
181 size_t current_load = 0;
182 if (current_sched.schedulers[static_cast<uint8_t>(SchedPolicy::kNormal)]) {
183 current_load =
184 current_sched.schedulers[static_cast<uint8_t>(SchedPolicy::kNormal)]
185 ->GetQueueSize();
186 }
187
188 // 寻找负载最高的核心
189 size_t max_load = 0;
190 size_t max_core = current_core;
191
192 for (size_t core_id = 0; core_id < SIMPLEKERNEL_MAX_CORE_COUNT; ++core_id) {
193 if (core_id == current_core) {
194 continue;
195 }
196 auto& other_sched = cpu_schedulers_[core_id];
197 if (other_sched.schedulers[static_cast<uint8_t>(SchedPolicy::kNormal)]) {
198 size_t load =
199 other_sched.schedulers[static_cast<uint8_t>(SchedPolicy::kNormal)]
200 ->GetQueueSize();
201 if (load > max_load) {
202 max_load = load;
203 max_core = core_id;
204 }
205 }
206 }
207
208 // 仅当差值 > 1 时才窃取（避免 ping-pong）
209 if (max_core == current_core || max_load <= current_load + 1) {
210 return;
211 }
212
213 // 按核心 ID 顺序获取锁，防止死锁
214 auto& source_sched = cpu_schedulers_[max_core];
215 size_t first_core = (current_core < max_core) ? current_core : max_core;
216 size_t second_core = (current_core < max_core) ? max_core : current_core;
217
218 LockGuard<SpinLock> lock_first(cpu_schedulers_[first_core].lock);
219 LockGuard<SpinLock> lock_second(cpu_schedulers_[second_core].lock);
220
221 // 重新检查（持锁后条件可能已变化）
222 auto* source_scheduler =
223 source_sched.schedulers[static_cast<uint8_t>(SchedPolicy::kNormal)].get();
224 auto* dest_scheduler =
225 current_sched.schedulers[static_cast<uint8_t>(SchedPolicy::kNormal)]
226 .get();
227
228 if (!source_scheduler || !dest_scheduler) {
229 return;
230 }
231
232 size_t source_load = source_scheduler->GetQueueSize();
233 size_t dest_load = dest_scheduler->GetQueueSize();
234
235 if (source_load <= dest_load + 1) {
236 return;
237 }
238
239 auto* stolen = source_scheduler->PickNext();
240 if (!stolen) {
241 return;
242 }
243
244 if (stolen->aux && stolen->aux->cpu_affinity.value() != UINT64_MAX &&
245 !(stolen->aux->cpu_affinity.value() & (1UL << current_core))) {
246 source_scheduler->Enqueue(stolen);
247 return;
248 }
249
250 dest_scheduler->Enqueue(stolen);
251 klog::Debug("Balance: Stole task '{}' (pid={}) from core {} to core {}",
252 stolen->name, stolen->pid, max_core, current_core);
253}

255auto TaskManager::ReapTask(TaskControlBlock* task) -> void {
256 if (!task) {
257 return;
258 }
259
260 // 确保任务处于僵尸或退出状态
261 if (task->GetStatus() != TaskStatus::kZombie &&
262 task->GetStatus() != TaskStatus::kExited) {
263 klog::Warn("ReapTask: Task {} is not in zombie/exited state", task->pid);
264 return;
265 }
266
267 // Capture pid before erase (unique_ptr deletes on erase)
268 Pid pid = task->pid;
269
270 // 从全局任务表中移除 (unique_ptr auto-deletes TCB)
271 {
272 LockGuard lock_guard{task_table_lock_};
273 task_table_.erase(pid);
274 }
275
276 klog::Debug("ReapTask: Task {} resources freed", pid);
277}

279auto TaskManager::ReparentChildren(TaskControlBlock* parent) -> void {
280 if (!parent) {
281 return;
282 }
283
284 LockGuard lock_guard{task_table_lock_};
285
286 // 遍历所有任务，找到父进程是当前任务的子进程
287 for (auto& [pid, task] : task_table_) {
288 if (task && task->aux->parent_pid == parent->pid) {
289 // 将子进程过继给 init 进程
290 task->aux->parent_pid = kInitPid;

292 task->pid, kInitPid);
