34 [[nodiscard]] __always_inline auto Lock() -> Expected<void> {
35 // 先 spin 获取锁——保证只有一个执行流通过此点。
36 // 重入在 spin 中通过 owner_core_ 检测（原子操作，多核安全）。
37 while (locked_.test_and_set(std::memory_order_acquire)) {
38 if (owner_core_.load(std::memory_order_acquire) ==
39 cpu_io::GetCurrentCoreId()) {
40 return std::unexpected(Error{ErrorCode::kSpinLockRecursiveLock});
41 }
42 cpu_io::Pause();
43 }
44 owner_core_.store(cpu_io::GetCurrentCoreId(), std::memory_order_release);
45
46 // 获取锁后操作 per-CPU lock_stack（此时中断已关，同核心无并发）
47 auto& stack = per_cpu::GetCurrentCore().lock_stack;
48 if (lock_level_ != lock_level::kUnclassified && stack.depth > 0) {
49 uint8_t top_level = stack.entries[stack.depth - 1].level;
50 if (top_level != lock_level::kUnclassified && lock_level_ < top_level) {
51 klog::RawPut("LOCK ORDER VIOLATION: acquiring '");
52 klog::RawPut(name);
53 klog::RawPut("' while holding '");
54 klog::RawPut(stack.entries[stack.depth - 1].lock->name);
55 klog::RawPut("'\n");
56 RawDumpStack();
57 while (true) {
58 cpu_io::Pause();
59 }
60 }
61 }
62 if (stack.depth < per_cpu::PerCpu::LockStack::kMaxDepth) {
63 stack.entries[stack.depth] = {this, lock_level_};
64 stack.depth++;
65 }
66
67 return {};
68 }

74 [[nodiscard]] __always_inline auto UnLock() -> Expected<void> {
75 if (owner_core_.load(std::memory_order_acquire) !=
76 cpu_io::GetCurrentCoreId()) {
77 return std::unexpected(Error{ErrorCode::kSpinLockNotOwned});
78 }
79
80 // 先弹栈（仍持有锁，per-CPU 数据安全），再释放锁
81 auto& stack = per_cpu::GetCurrentCore().lock_stack;
82 if (stack.depth > 0 && stack.entries[stack.depth - 1].lock == this) {
83 stack.depth--;
84 }
85
86 owner_core_.store(std::numeric_limits<size_t>::max(),
87 std::memory_order_release);
88 locked_.clear(std::memory_order_release);
89 return {};
90 }
