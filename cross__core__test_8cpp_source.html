24// ===========================================================================
25// test_cross_core_wakeup
26//
27// Two tasks pinned to different cores share a Mutex.
28// Task A (core 0) acquires it, sleeps 200ms, releases.
29// Task B (core 1) blocks on Lock(), wakes up when A releases.
30// Verify B acquired the mutex after A released it.
31// ===========================================================================
32
33struct WakeupArgs {
34 Mutex* mtx;
35 std::atomic<int>* phase; // 0:init, 1:A locked, 2:A unlocked, 3:B locked
36};
37
38void wakeup_task_a(void* arg) {
39 auto* ctx = reinterpret_cast<WakeupArgs*>(arg);
40
41 (void)ctx->mtx->Lock();
42 ctx->phase->store(1, std::memory_order_release);
43 klog::Info("[CrossCore Wakeup] A: locked mutex, sleeping 200ms");
44
45 (void)sys_sleep(200);
46
47 (void)ctx->mtx->UnLock();
48 ctx->phase->store(2, std::memory_order_release);
49 klog::Info("[CrossCore Wakeup] A: unlocked mutex");
50
51 sys_exit(0);
52}
53
54void wakeup_task_b(void* arg) {
55 auto* ctx = reinterpret_cast<WakeupArgs*>(arg);
56
57 // Wait until A has locked the mutex before trying
58 int timeout = 100;
59 while (timeout > 0 && ctx->phase->load(std::memory_order_acquire) < 1) {
60 (void)sys_sleep(10);
61 timeout--;
62 }
63
64 klog::Info("[CrossCore Wakeup] B: attempting Lock() (should block)");
65 (void)ctx->mtx->Lock();
66
67 // By the time B gets here, A must have released (phase >= 2)
68 int phase_val = ctx->phase->load(std::memory_order_acquire);
69 if (phase_val < 2) {
70 klog::Err(
71 "[CrossCore Wakeup] B: acquired mutex before A released! "
72 "phase={}",
73 phase_val);
74 }
75 ctx->phase->store(3, std::memory_order_release);
76 klog::Info("[CrossCore Wakeup] B: locked mutex after A released");
77
78 (void)ctx->mtx->UnLock();
79 sys_exit(0);
80}

148void exit_wait_parent(void* arg) {
149 auto* ctx = reinterpret_cast<ExitWaitArgs*>(arg);
150
151 auto& task_mgr = TaskManagerSingleton::instance();
152 auto* current = task_mgr.GetCurrentTask();
153
154 // Create child pinned to core 1
155 auto child = kstd::make_unique<TaskControlBlock>("CCExitChild", 10,
156 exit_wait_child, nullptr);
157 child->aux->parent_pid = current->pid;
158 child->aux->pgid = current->aux->pgid;
159 child->aux->cpu_affinity = (1UL << 1);
160
161 // Save raw pointer: pid is assigned inside AddTask() by AllocatePid(),
162 // so we must read it *after* the call (unique_ptr is moved).
163 auto* child_ptr = child.get();
164 task_mgr.AddTask(std::move(child));
165 Pid child_pid = child_ptr->pid;
166
167 klog::Info(
168 "[CrossCore ExitWait] Parent: created child pid={}, calling "
169 "Wait()",
170 child_pid);
171
172 int status = 0;
173 auto wait_result = task_mgr.Wait(static_cast<Pid>(-1), &status, false, false);
174
175 if (wait_result.has_value() && wait_result.value() == child_pid) {
176 ctx->result_code->store(status, std::memory_order_release);
177 klog::Info("[CrossCore ExitWait] Parent: child exited with code {}",
178 status);
179 } else {
180 ctx->result_code->store(-1, std::memory_order_release);
181 klog::Err("[CrossCore ExitWait] Parent: Wait() failed or wrong pid");
182 }
183
184 ctx->result_ready->store(1, std::memory_order_release);
185 sys_exit(0);
186}

232// ===========================================================================
233// test_cross_core_mutex_counter
234//
235// 4 tasks (2 on core 0, 2 on core 1) each increment a shared counter
236// 100 times under mutex protection. Final counter must be 400.
237// ===========================================================================
238
239struct MutexCounterArgs {
240 Mutex* mtx;
241 std::atomic<int>* counter;
242 int iterations;
243};
244
245void mutex_counter_worker(void* arg) {
246 auto* ctx = reinterpret_cast<MutexCounterArgs*>(arg);
247
248 for (int i = 0; i < ctx->iterations; ++i) {
249 (void)ctx->mtx->Lock();
250 int val = ctx->counter->load(std::memory_order_relaxed);
251 ctx->counter->store(val + 1, std::memory_order_relaxed);
252 (void)ctx->mtx->UnLock();
253 }
254
255 sys_exit(0);
256}
