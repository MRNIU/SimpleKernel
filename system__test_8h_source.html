209inline void QemuExit([[maybe_unused]] bool success) {
210#if defined(__riscv)
211 // sifive_test device (virt machine 默认存在，地址 0x100000)
212 // 0x5555 = FINISHER_PASS, 0x3333 = FINISHER_FAIL
213 constexpr uint64_t kSifiveTestAddr = 0x100000;
214 constexpr size_t kSifiveTestSize = 0x1000;
215 (void)VirtualMemorySingleton::instance().MapMMIO(kSifiveTestAddr,
216 kSifiveTestSize);
217 volatile auto* finisher =
218 reinterpret_cast<volatile uint32_t*>(kSifiveTestAddr);
219 *finisher = success ? 0x5555 : 0x3333;
220#elif defined(__aarch64__)
221 // PSCI SYSTEM_OFF，通过 SMC 调用 ATF (EL3)
222 cpu_io::SecureMonitorCall(cpu_io::psci::kSYSTEM_OFF, 0, 0, 0, 0, 0, 0, 0);
223#endif
224 __builtin_unreachable();
225}
